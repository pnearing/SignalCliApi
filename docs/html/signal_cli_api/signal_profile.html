197    def __save__(self) -> bool:
198        """
199        Save this profile to disk.
200        :return: bool: True if successfully saved, False if not.
201        :raises RuntimeError: On error opening the profile json file.
202        """
203        # Setup logging:
204        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.__save__.__name__)
205        # Checks:
206        if not self._is_account_profile:
207            warning_message: str = "This isn't the account profile, not saving."
208            logger.warning(warning_message)
209            return False
210        if self._profile_file_path is None:
211            warning_message: str = "'self._profile_file_path' is None, not saving."
212            logger.warning(warning_message)
213            return False
214        # Create json string to save:

217        # Open the file:

228    def __load__(self) -> bool:
229        """
230        Load this profile from disk.
231        :return: bool: True this was successfully loaded, False it was not.
232        """
233        # Setup logging:
234        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.__load__.__name__)
235
236        # Do checks:
237        if not self._is_account_profile:
238            warning_message: str = "This is not the account profile, not loading."

242            warning_message: str = "'self._profile_file_path' is None, not loading."
243            logger.warning(warning_message)
244            return False
245        # Try to open file:
246        try:
247            with open(self._profile_file_path, 'r', encoding='utf-8') as file_handle:
248                profile_dict: dict[str, object] = json.loads(file_handle.read())
249        except (OSError, FileNotFoundError, PermissionError) as e:
250            error_message: str = (f"Couldn't open file '{self._profile_file_path}' for "

253            raise RuntimeError(error_message) from e
254        except json.JSONDecodeError as e:
255            error_message: str = f"Couldn't load json from '{self._profile_file_path}': {e.msg}"
256            logger.critical("Raising InvalidDataFile(%s).", error_message)
257            raise InvalidDataFile(error_message, e, self._profile_file_path) from e
258        # Load from dict:
259        self.__from_dict__(profile_dict)
260        return True

265    def __find_avatar__(self) -> bool:
266        """
267        Find the avatar file for this profile.
268        :return: bool: True if the avatar was found, False if not.
269        """
270        # Setup logging:
271        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.__find_avatar__.__name__)
272
273        # Check if avatar already exists:
274        if self.avatar is not None:
275            if os.path.exists(self.avatar):
276                return True
277            warning_message: str = ("Current avatar points to non-existent file, searching "
278                                    "for new avatar.")
279            logger.warning(warning_message)
280            self.avatar = None
281
282        # Try profile avatar:
283        avatar_filename = 'profile-' + self._contact_id
284        avatar_file_path = os.path.join(self._config_path, 'avatars', avatar_filename)
285        if os.path.exists(avatar_file_path):
286            self.avatar = avatar_file_path
287            return True
288
289        # Try contact avatar:
290        avatar_filename = 'contact-' + self._contact_id
291        avatar_file_path = os.path.join(self._config_path, 'avatars', avatar_filename)
292        if os.path.exists(avatar_file_path):
293            self.avatar = avatar_file_path
294            return True
295        # Avatar was not found:
296        return False
