 327    def __parse_reaction__(self, reaction: SignalReaction) -> bool:
 328        """
 329        Parse a Reaction message.
 330        :param reaction: SignalReaction: The reaction to parse.
 331        :return: bool: True if the reaction was parsed, False if not.
 332        """
 333        # Setup logging:
 334        logger: logging.Logger = logging.getLogger(__name__ + '.' +
 335                                                   self.__parse_reaction__.__name__)
 336
 337        # Get the messages from the recipients:
 338        search_messages: list[SignalMessage] = []
 339        if reaction.recipient_type == RecipientTypes.CONTACT:
 340            messages = self.get_by_sender(reaction.recipient)
 341            search_messages.extend(messages)
 342        elif reaction.recipient_type == RecipientTypes.GROUP:
 343            messages = self.get_by_recipient(reaction.recipient)
 344            search_messages.extend(messages)
 345        else:
 346            # Invalid recipient type:
 347            error_message = "Invalid reaction cannot parse."
 348            logger.critical("Raising RuntimeError(%s).", error_message)
 349            raise RuntimeError(error_message)
 350        # Find the message that was reacted to:
 351        reacted_message: Optional[SignalSentMessage | SignalReceivedMessage | SignalMessage] = None
 352        for message in search_messages:
 353            if message.sender == reaction.target_author:
 354                if message.timestamp == reaction.target_timestamp:
 355                    reacted_message = message
 356        # If the message isn't in history, do nothing:
 357        if reacted_message is None:
 358            return False
 359        # Have the message add / change / remove the reaction:
 360        reacted_message.reactions.__parse__(reaction)
 361        self.__save__()  # save the results.
 362        return True

 364    def __parse_receipt__(self, receipt: SignalReceipt) -> None:
 365        """
 366        Parse a receipt message.
 367        :param receipt: SignalReceipt: The receipt message to parse.
 368        :return: None
 369        """
 370        # Mark seen:
 371        receipt.sender.__seen__(receipt.timestamp)
 372        receipt.device.__seen__(receipt.timestamp)
 373        receipt.recipient.__seen__(receipt.timestamp)
 374        # Build a list of unparsed receipts
 375        receipts = [receipt, *self._unparsed_receipts]
 376        self._unparsed_receipts = []
 377        should_save: bool = False
 378        for _receipt in receipts:
 379            # Parse receipts:
 380            receipt_parsed: bool = False
 381            for message in self.get_sent_messages():
 382                for timestamp in _receipt.timestamps:
 383                    if message.timestamp == timestamp:
 384                        message.__parse_receipt__(_receipt)
 385                        receipt_parsed = True
 386            if receipt_parsed:
 387                should_save = True
 388            else:
 389                self._unparsed_receipts.append(_receipt)
 390        if should_save:
 391            self.__save__()

 393    def __parse_read_message_sync__(self, sync_message: SignalSyncMessage) -> None:
 394        """
 395        Parse a read message sync message.
 396        :param sync_message: SyncMessage: The sync message to parse.
 397        :return: None
 398        """
 399        # Mark sync_message seen:
 400        sync_message.sender.__seen__(sync_message.timestamp)
 401        sync_message.device.__seen__(sync_message.timestamp)
 402        sync_message.recipient.__seen__(sync_message.timestamp)
 403
 404        # Parse the read message sync message:
 405        should_save: bool = False
 406        for contact, timestamp in sync_message.read_messages:
 407            search_messages = self.get_by_sender(contact)
 408            for message in search_messages:
 409                if message.timestamp == timestamp:
 410                    if not message.is_read:
 411                        should_save = True
 412                        if isinstance(message, SignalReceivedMessage):
 413                            message.mark_read(when=sync_message.timestamp, send_receipt=False)
 414                        else:
 415                            message.mark_read(when=sync_message.timestamp)
 416        if should_save:
 417            self.__save__()

 443        """

 564        :returns: list[SentMessage|ReceivedMessage]: The conversation, or an empty list if not
 565        found.
 566        :raises: TypeError: If target is not a SignalContact or SignalGroup object.
 567        """
 568        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.get_conversation.__name__)
 569        return_messages = []
 570        if isinstance(target, SignalContact):
 571            self_contact = self._contacts.get_self()
 572            for message in self.messages:
 573                if message.sender == self_contact and message.recipient == target:
 574                    return_messages.append(message)
 575                elif message.sender == target and message.recipient == self_contact:
 576                    return_messages.append(message)
 577        elif isinstance(target, SignalGroup):
 578            for message in self.messages:
 579                if message.recipient.get_id() == target.get_id():
 580                    return_messages.append(message)
 581                # if message.recipient == target:
 582                #     return_messages.append(message)
 583        else:
 584            logger.critical("Raising TypeError:")
 585            __type_error__("target", "SignalContact | SignalGroup", target)
 586        # Apply filters:
 587        if message_filter == MessageFilter.NONE:
 588            return return_messages
 589        if message_filter & MessageFilter.READ:
 590            return_messages = [message for message in return_messages if message.is_read is True]
 591        elif message_filter & MessageFilter.NOT_READ:
 592            return_messages = [message for message in return_messages if message.is_read is False]
 593        if message_filter & MessageFilter.VIEWED:
 594            return_messages = [message for message in return_messages if message.is_viewed is True]
 595        elif message_filter & MessageFilter.NOT_VIEWED:
 596            return_messages = [message for message in return_messages if message.is_viewed is False]
 597        if message_filter & MessageFilter.DELIVERED:
 598            return_messages = [message for message in return_messages if
 599                               message.is_delivered is True]
 600        elif message_filter & MessageFilter.NOT_DELIVERED:
 601            return_messages = [message for message in return_messages if
 602                               message.is_delivered is False]
 603        return return_messages

 328    def __parse_reaction__(self, reaction: SignalReaction) -> bool:
 329        """
 330        Parse a Reaction message.
 331        :param reaction: SignalReaction: The reaction to parse.
 332        :return: bool: True if the reaction was parsed, False if not.
 333        """
 334        # Setup logging:
 335        logger: logging.Logger = logging.getLogger(__name__ + '.' +
 336                                                   self.__parse_reaction__.__name__)
 337
 338        # Get the messages from the recipients:
 339        search_messages: list[SignalMessage] = []
 340        if reaction.recipient_type == RecipientTypes.CONTACT:
 341            messages = self.get_by_sender(reaction.recipient)
 342            search_messages.extend(messages)
 343        elif reaction.recipient_type == RecipientTypes.GROUP:
 344            messages = self.get_by_recipient(reaction.recipient)
 345            search_messages.extend(messages)
 346        else:
 347            # Invalid recipient type:
 348            error_message = "Invalid reaction cannot parse."
 349            logger.critical("Raising RuntimeError(%s).", error_message)
 350            raise RuntimeError(error_message)
 351        # Find the message that was reacted to:
 352        reacted_message: Optional[SignalSentMessage | SignalReceivedMessage | SignalMessage] = None
 353        for message in search_messages:
 354            if message.sender == reaction.target_author:
 355                if message.timestamp == reaction.target_timestamp:
 356                    reacted_message = message
 357        # If the message isn't in history, do nothing:
 358        if reacted_message is None:
 359            return False
 360        # Have the message add / change / remove the reaction:
 361        reacted_message.reactions.__parse__(reaction)
 362        self.__save__()  # save the results.
 363        return True

 365    def __parse_receipt__(self, receipt: SignalReceipt) -> None:
 366        """
 367        Parse a receipt message.
 368        :param receipt: SignalReceipt: The receipt message to parse.
 369        :return: None
 370        """
 371        # Mark seen:
 372        receipt.sender.__seen__(receipt.timestamp)
 373        receipt.device.__seen__(receipt.timestamp)
 374        receipt.recipient.__seen__(receipt.timestamp)
 375        # Build a list of unparsed receipts
 376        receipts = [receipt, *self._unparsed_receipts]
 377        self._unparsed_receipts = []
 378        should_save: bool = False
 379        for _receipt in receipts:
 380            # Parse receipts:
 381            receipt_parsed: bool = False
 382            for message in self.get_sent_messages():
 383                for timestamp in _receipt.timestamps:
 384                    if message.timestamp == timestamp:
 385                        message.__parse_receipt__(_receipt)
 386                        receipt_parsed = True
 387            if receipt_parsed:
 388                should_save = True
 389            else:
 390                self._unparsed_receipts.append(_receipt)
 391        if should_save:
 392            self.__save__()

 414                            message.mark_read(when=sync_message.timestamp, send_receipt=False)

 416                            message.mark_read(when=sync_message.timestamp)

 556    def get_conversation(self,
 557                         target: SignalContact | SignalGroup,
 558                         message_filter: int = MessageFilter.NONE,
 559                         ) -> list[SignalMessage]:
 560        """
 561        Get a conversation, given a target contact or group.  Returns a list of messages either
 562        sent to or received from a contact or group.
 563        :param target: SignalContact | SignalGroup: The contact or group to search for.
 564        :param message_filter: int: The filter. Use signalCommon.MessageFilter flag values.
 565        :returns: list[SentMessage|ReceivedMessage]: The conversation, or an empty list if not
 566        found.
 567        :raises: TypeError: If target is not a SignalContact or SignalGroup object.
 568        """
 569        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.get_conversation.__name__)
 570        return_messages = []
 571        if isinstance(target, SignalContact):
 572            self_contact = self._contacts.get_self()
 573            for message in self.messages:
 574                if message.sender == self_contact and message.recipient == target:
 575                    return_messages.append(message)
 576                elif message.sender == target and message.recipient == self_contact:
 577                    return_messages.append(message)
 578        elif isinstance(target, SignalGroup):
 579            for message in self.messages:
 580                if message.recipient.get_id() == target.get_id():
 581                    return_messages.append(message)
 582                # if message.recipient == target:
 583                #     return_messages.append(message)
 584        else:
 585            logger.critical("Raising TypeError:")
 586            __type_error__("target", "SignalContact | SignalGroup", target)
 587        # Apply filters:
 588        if message_filter == MessageFilter.NONE:
 589            return return_messages
 590        if message_filter & MessageFilter.READ:
 591            return_messages = [message for message in return_messages if message.is_read is True]
 592        elif message_filter & MessageFilter.NOT_READ:
 593            return_messages = [message for message in return_messages if message.is_read is False]
 594        if message_filter & MessageFilter.VIEWED:
 595            return_messages = [message for message in return_messages if message.is_viewed is True]
 596        elif message_filter & MessageFilter.NOT_VIEWED:
 597            return_messages = [message for message in return_messages if message.is_viewed is False]
 598        if message_filter & MessageFilter.DELIVERED:
