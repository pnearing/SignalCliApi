323    def __send_receipt__(self, receipt_type: ReceiptTypes) -> tuple[bool, SignalTimestamp | str]:
324        """
325        Send a receipt using signal.
326        :param receipt_type: ReceiptTypes: The type of receipt to send; Either ReceiptTypes.READ
327            or ReceiptTypes.VIEWED.
328        :return: tuple[bool, str | SignalTimestamp]: The first element is True or False for
329            success or failure. The second element is either the SignalTimestamp object of the
330            receipts' 'when' on success, or an error message, stating what went wrong.
331        :raises RuntimeError: On invalid receipt type.
332        :raises CommunicationsError: On error communicating with signal.
333        :raises InvalidServerResponse: On error loading signal JSON.
334        :raises SignalError: On error sent by signal.
335        """
336        # Setup logging:
337        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.__send_receipt__.__name__)
338        # Parse receipt type:
339        type_string: str
340        if receipt_type == ReceiptTypes.READ:
341            type_string = 'read'
342        elif receipt_type == ReceiptTypes.VIEWED:
343            type_string = 'viewed'
344        else:
345            error_message: str = f"Can't send this type of receipt: {str(receipt_type)}."
346            logger.critical("Raising RuntimeError(%s).", error_message)
347            raise RuntimeError(error_message)
348
349        # Create send receipt command object and json command string.
350        send_receipt_command_obj = {
351            "jsonrpc": "2.0",
352            "id": 0,
353            "method": "sendReceipt",
354            "params": {
355                "account": self._account_id,
356                "recipient": self.sender.get_id(),
357                "type": type_string,
358                "targetTimestamp": self.timestamp.timestamp,
359            }
360        }
361        json_command_str: str = json.dumps(send_receipt_command_obj) + '\n'
362
363        # Communicate with signal:
364        __socket_send__(self._command_socket, json_command_str)
365        response_str: str = __socket_receive_blocking__(self._command_socket)
366        response_obj: dict[str, Any] = __parse_signal_response__(response_str)
367
368        # Check for error:
369        error_occurred, signal_code, signal_message = __check_response_for_error__(response_obj, [])
370        if error_occurred:
371            error_message: str = (f"signal error while sending receipt. "
372                                  f"Code: {signal_code}, Message: {signal_message}")
373            logger.warning(error_message)
374            return False, error_message
375
376        # Result is a dict:
377        result_obj = response_obj['result']
378        when = SignalTimestamp(timestamp=result_obj['timestamp'])
379        # Parse results:
380        for result in result_obj['results']:
381            if result['type'] != 'SUCCESS':
382                warning_message: str = (f"While sending result['type'] != 'SUCCESS'. result['type']"
383                                        f"== {result['type']}")
384                logger.warning(warning_message)
385            else:
386                recipient: dict[str, str] = result['recipientAddress']
387                _, contact = self._contacts.__get_or_add__(number=recipient['number'],
388                                                           uuid=recipient['uuid'])
389                contact.__seen__(when)
390        return True, when

405    def __check_invite__(self) -> bool:
406        """
407        Check if this is a group invite, it's an invitation if it's a group message without a body,

419        """

435    def mark_read(self, when: SignalTimestamp = None, send_receipt: bool = True) -> None:
436        """
437        Mark the message as read.
438        :param when: SignalTimestamp: When the message was read; If this is None, NOW is used.
439        :param send_receipt: bool: Send the read receipt; If this is True, 'when' is ignored.
440        :returns: None
441        :raises TypeError: If when not a SignalTimestamp object, raised by super(), or if
442        send_receipt is not a bool.
443        :raises RuntimeError: On error sending receipt.
444        """
445        # Setup logging:
446        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.mark_read.__name__)
447        # Type check send_receipt, when is type checked by super.
448        if not isinstance(send_receipt, bool):
449            logger.critical("Raising TypeError:")
450            __type_error__("send_receipt", "bool", send_receipt)
451        # Send the read receipt if requested:
452        if send_receipt:
453            is_success, results = self.__send_receipt__(ReceiptTypes.READ)
454            if not is_success:
455                error_message: str = f"failed to send read receipt: {results}"
456                logger.critical("Raising RuntimeError(%s).", error_message)
457                raise RuntimeError(error_message)
458            time_read = results
459        else:
460            time_read = when
461        # Set expiry and run super()
462        self.__set_expiry__(time_read)
463        super().mark_read(time_read)

324    def __send_receipt__(self, receipt_type: ReceiptTypes) -> tuple[bool, SignalTimestamp | str]:
325        """
326        Send a receipt using signal.
327        :param receipt_type: ReceiptTypes: The type of receipt to send; Either ReceiptTypes.READ
328            or ReceiptTypes.VIEWED.
329        :return: tuple[bool, str | SignalTimestamp]: The first element is True or False for
330            success or failure. The second element is either the SignalTimestamp object of the
331            receipts' 'when' on success, or an error message, stating what went wrong.
332        :raises RuntimeError: On invalid receipt type.
333        :raises CommunicationsError: On error communicating with signal.
334        :raises InvalidServerResponse: On error loading signal JSON.
335        :raises SignalError: On error sent by signal.
336        """
337        # Setup logging:
338        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.__send_receipt__.__name__)
339        # Parse receipt type:
340        type_string: str
341        if receipt_type == ReceiptTypes.READ:
342            type_string = 'read'
343        elif receipt_type == ReceiptTypes.VIEWED:
344            type_string = 'viewed'
345        else:
346            error_message: str = f"Can't send this type of receipt: {str(receipt_type)}."
347            logger.critical("Raising RuntimeError(%s).", error_message)
348            raise RuntimeError(error_message)
349
350        # Create send receipt command object and json command string.
351        send_receipt_command_obj = {
352            "jsonrpc": "2.0",
353            "id": 0,
354            "method": "sendReceipt",
355            "params": {
356                "account": self._account_id,
357                "recipient": self.sender.get_id(),
358                "type": type_string,
359                "targetTimestamp": self.timestamp.timestamp,
360            }
361        }
362        json_command_str: str = json.dumps(send_receipt_command_obj) + '\n'
363
364        # Communicate with signal:
365        __socket_send__(self._command_socket, json_command_str)
366        response_str: str = __socket_receive_blocking__(self._command_socket)
367        response_obj: dict[str, Any] = __parse_signal_response__(response_str)
368
369        # Check for error:
370        error_occurred, signal_code, signal_message = __check_response_for_error__(response_obj, [])
371        if error_occurred:
372            error_message: str = (f"signal error while sending receipt. "
373                                  f"Code: {signal_code}, Message: {signal_message}")
374            logger.warning(error_message)
375            return False, error_message
376
377        # Result is a dict:
378        result_obj = response_obj['result']
379        when = SignalTimestamp(timestamp=result_obj['timestamp'])
380        # Parse results:
381        for result in result_obj['results']:
382            if result['type'] != 'SUCCESS':
383                warning_message: str = (f"While sending result['type'] != 'SUCCESS'. result['type']"
384                                        f"== {result['type']}")
385                logger.warning(warning_message)
386            else:
387                recipient: dict[str, str] = result['recipientAddress']
388                _, contact = self._contacts.__get_or_add__(number=recipient['number'],
389                                                           uuid=recipient['uuid'])
390                contact.__seen__(when)
391        return True, when
