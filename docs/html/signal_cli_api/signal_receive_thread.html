213    def __call_callback__(self,
214                          callback: Optional[tuple[Callable, Optional[list[Any]]]],
215                          account: SignalAccount,
216                          message: SignalMessage,
217                          ) -> Optional[bool]:
218        """
219        Execute a callback and return True for stopping reception, False for do not stop reception;
220        The order of priority of return values is all messages callback, specified callback. So if
221        all messages returns None, then the return value of the specified callback is returned.
222        :param callback: Optional[tuple[Callable, Optional[list[Any]]]]: The callback to call,
223            and any parameters to pass to it, if None the callback is not executed.
224        :param account: SignalAccount: The account we're receiving for.
225        :param message: SignalMessage: The message we've received.
226        :return: Optional[bool]: If True is returned, the callback stops the reception thread, If
227            False or None are  returned, then reception continues.
228        """
229        # Setup logging:
230        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.__call_callback__.__name__)
231        # Call specified call back:
232        logger.debug("Calling specific callback for: %s", str(type(message)))
233        cb_return_value: Optional[bool] = __run_callback__(callback, account, message)
234        # Call all messages callback:
235        logger.debug("Calling all message callback.")
236        all_return_value: Optional[bool] = __run_callback__(self._all_msg_cb, account, message)
237        # Determine return value:
238        if cb_return_value is None:
239            logger.debug("Returning all message callback return value.")
240            return all_return_value
241        logger.debug("Returning specific message callback return value.")
242        return cb_return_value

352    def __parse_sync_message__(self, envelope_dict: dict[str, Any]) -> Optional[bool]:
353        """
354        Parse a sync message.
355        :param envelope_dict: dict[str, Any]: The incoming message.
356        :return: Optional[bool]: The return value of the callback.
357        """
358        # Setup logging:
359        logger: logging.Logger = logging.getLogger(__name__ + '.'
360                                                   + self.__parse_sync_message__.__name__)
361
362        # Check to see if this is an empty sync message:
363        if envelope_dict['syncMessage'] == {}:
364            logger.warning("Got empty sync message, skipping.")
365            return None
366        # Create the SignalSyncMessage object:
367        message = SignalSyncMessage(
368            command_socket=self._command_socket, account_id=self._account.number,
369            config_path=self._config_path, contacts=self._account.contacts,
370            groups=self._account.groups, devices=self._account.devices,
371            this_device=self._account.device, sticker_packs=self._sticker_packs,
372            raw_message=envelope_dict
373        )
374        if message.sync_type in (SyncTypes.READ_MESSAGES, SyncTypes.SENT_MESSAGES,
375                                 SyncTypes.SENT_REACTION):
376            self._account.messages.__parse_sync_message__(message)
377        elif message.sync_type == SyncTypes.CONTACTS:
378            self._account.contacts.__sync__()
379        elif message.sync_type == SyncTypes.GROUPS:
380            self._account.groups.__sync__()
381        elif message.sync_type == SyncTypes.BLOCKS:
382            self._account.contacts.__parse_sync_message__(message)
383            self._account.groups.__parse_sync_message__(message)
384        elif message.sync_type == SyncTypes.SENT_MESSAGES:
385            self._account.messages.__parse_sync_message__(message)
386        else:
387            error_message: str = f"Unhandled sync type: {message.sync_type}"
388            logger.critical("Raising RuntimeError(%s).", error_message)
389            raise RuntimeError(error_message)
390        # Append the message to messages:
391        self._account.messages.append(message)
392        # Call sync message callback:
393        return self.__call_callback__(self._sync_msg_cb, self._account, message)

448    def run(self) -> None:
449        """
450        Thread override.
451        :return: None
452        """
453        # Setup logging:
454        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.run.__name__)
455
456        # Do send sync request if we're not the primary device.
457        if self._account.device_id != 1:
458            # Create sync request object and json command string:
459            sync_request_command_obj: dict[str, Any] = {
460                "jsonrpc": "2.0",
461                "id": 10,
462                "method": "sendSyncRequest",
463                "params": {
464                    "account": self._account.number,
465                }
466            }
467            json_command_str: str = json.dumps(sync_request_command_obj) + '\n'
468
469            # Communicate with Signal:
470            __socket_send__(self._receive_socket, json_command_str)
471            response_str = __socket_receive_blocking__(self._receive_socket)
472            response_obj: dict[str, Any] = __parse_signal_response__(response_str)
473            # There are no non fatal errors during reception:
474            __check_response_for_error__(response_obj, [])
475
476        # Create receive object and json command string:
477        start_receive_command_object: dict[str, Any] = {
478            "jsonrpc": "2.0",
479            "id": 1,
480            "method": "subscribeReceive",
481            "params": {
482                "account": self._account.number,
483            }
484        }
485        json_command_str: str = json.dumps(start_receive_command_object) + '\n'
486
487        # Communicate start receive with signal:
488        __socket_send__(self._receive_socket, json_command_str)
489        response_str = __socket_receive_blocking__(self._receive_socket)
490        response_obj: dict[str, Any] = __parse_signal_response__(response_str)
491        error_occurred, signal_code, signal_message = __check_response_for_error__(response_obj, [])
492        if error_occurred:
493            error_message: str = (f"Signal error while trying to start receiving. "
494                                  f"Code {signal_code}, Message: {signal_message}")
495            logger.critical("Raising RuntimeError(%s).", error_message)
496            raise RuntimeError(error_message)
497
498        # Set subscription ID, and start receiving:
499        self._subscription_id = response_obj['result']
500        self._receiving = True
501        # START RECEIVE LOOP:
502        while self._receiving:
503            try:
504                response_str: Optional[str] = __socket_receive_non_blocking__(self._receive_socket,
505                                                                              0.01)
506            except CommunicationsError as e:
507                if self._receiving is False:
508                    break
509                raise e
510            if response_str is None:
511                continue
512            # Delay processing until messages are finished sending:
513            if self._account.messages.sending:
514                logger.debug("Message received while sending a message, delaying processing "
515                             "until complete.")
516            while self._account.messages.sending:
517                pass
518
519            # Create msg object, and check the incoming message for an error,
520            # NOTE: There are no non-fatal errors during reception:
521            message_obj: dict[str, Any] = __parse_signal_response__(response_str)
522            __check_response_for_error__(response_obj, [])
523
524            # Make sure there is a method in the response:
525            if 'method' not in message_obj.keys():
526                logger.warning("Message received with no method.")
527                logger.debug("message_obj = %s", str(message_obj))
528                continue
529
530            # Make sure that method is 'receive':
531            if message_obj['method'] != 'receive':
532                logger.warning("Message received with method other than 'receive': method = %s",
533                               message_obj['method'])
534                logger.debug('message_obj = %s', str(message_obj))
535                continue
536
537            # Make sure there are 'params' in the response:
538            if 'params' not in message_obj.keys():
539                logger.warning("Message has no 'params'.")
540                logger.debug("message_obj = %s", str(message_obj))
541                continue
542
543            # Make sure there are 'result' in the 'params':
544            if 'result' not in message_obj['params'].keys():
545                logger.warning("Message doesn't have a result.")
546                logger.debug("message_obj = %s", str(message_obj))
547                continue
548
549            # Make sure there is an 'envelope' in the message 'result':
550            if 'envelope' not in message_obj['params']['result'].keys():
551                logger.warning("Message with no envelope received.")
552                logger.debug("message_obj = %s", str(message_obj))
553                continue
554
555            # Grab the envelope dict and type return_value:
556            envelope_dict: dict = message_obj['params']['result']['envelope']
557            return_value: Optional[bool]
558            ##########################
559            # Data Message:
560            ##########################
561            if 'dataMessage' in envelope_dict.keys():
562                return_value = self.__parse_data_message__(envelope_dict)
563                if return_value is True:
564                    break  # Stop receiving.
565            ##########################
566            # Receipt Message:
567            ##########################
568            elif 'receiptMessage' in envelope_dict.keys():
569                return_value = self.__parse_receipt_message__(envelope_dict)
570                if return_value is True:
571                    break  # Stop receiving.
572            #############################
573            # Sync Message:
574            #############################
575            elif 'syncMessage' in envelope_dict.keys():
576                return_value = self.__parse_sync_message__(envelope_dict)
577                if return_value is True:
578                    break  # Stop receiving.
579            ###############################
580            # Typing Message:
581            ###############################
582            elif 'typingMessage' in envelope_dict.keys():
583                return_value = self.__parse_typing_message__(envelope_dict)
584                if return_value is True:
585                    break  # Stop receiving.
586            ###############################
587            # Story Message:
588            ###############################
589            elif 'storyMessage' in envelope_dict.keys():
590                return_value = self.__parse_story_message__(envelope_dict)
591                if return_value is True:
592                    break  # Stop receiving.
593            ##############################
594            # Call message:
595            ##############################
596            elif 'callMessage' in envelope_dict.keys():
597                return_value = self.__parse_call_message__(envelope_dict)
598                if return_value is True:
599                    break  # Stop receiving.
600            ##############################
601            # Unrecognized message:
602            ##############################
603            else:
604                logger.warning("Unrecognized incoming envelope. Perhaps a payment message.")
605                logger.debug("envelope_dict.keys() = %s", str(envelope_dict.keys()))
606                logger.debug("envelope_dict = %s", str(envelope_dict))
607                continue
608
609            ###############################
610            # Check for expired messages:
611            ###############################
612            # self._account.messages.__check_expiries__()
613            if self._do_expunge:
614                self._account.messages.do_expunge()

360        logger: logging.Logger = logging.getLogger(__name__ + '.'
