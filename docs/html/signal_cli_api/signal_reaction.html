228    def send(self) -> tuple[bool, str]:
229        """
230        Send the reaction.
231        :returns: tuple[bool, str]: True/False for sent status, string for an error message if
232            False, or "SUCCESS" if True.
233        """
234        # Setup logging:
235        logger: logging.Logger = logging.getLogger(__name__ + '.' + self.send.__name__)
236
237        # Check if this was already sent.
238        if self.is_sent:
239            error_message: str = "reaction already sent."
240            logger.critical("Raising RuntimeError(%s).", error_message)
241            raise RuntimeError(error_message)
242
243        # Create reaction command object and json command string:
244        send_reaction_command_obj = {
245            "jsonrpc": "2.0",
246            "id": 10,
247            "method": "sendReaction",
248            "params": {
249                "account": self._account_id,
250                "emoji": self.emoji,
251                "targetAuthor": self.target_author.get_id(),
252                "targetTimestamp": self.target_timestamp.timestamp,
253            }
254        }
255        # Add recipient to the message:
256        if self.recipient_type == RecipientTypes.CONTACT:
257            send_reaction_command_obj['params']['recipient'] = self.sender.get_id()
258        elif self.recipient_type == RecipientTypes.GROUP:
259            send_reaction_command_obj['params']['groupId'] = self.recipient.get_id()
260        else:
261            raise ValueError(f"recipient type = {str(self.recipient_type)}")
262
263        # Create the JSON command string:
264        json_command_str: str = json.dumps(send_reaction_command_obj) + '\n'
265
266        # Communicate with signal:
267        __socket_send__(self._command_socket, json_command_str)
268        response_str: str = __socket_receive_blocking__(self._command_socket)
269        response_obj: dict[str, Any] = __parse_signal_response__(response_str)
270
271        # Check for error:
272        error_occurred, error_code, error_message = __check_response_for_error__(response_obj, [])
273        if error_occurred:
274            error_message: str = (f"signal error while sending reaction. "
275                                  f"Code: {error_code}, Message: {error_message}")
276            return False, error_message
277
278        # Parse Response:
279        result_obj: dict[str, Any] = response_obj['result']
280        self.timestamp = SignalTimestamp(timestamp=result_obj['timestamp'])
281        # Check for delivery error:
282        if result_obj['results'][0]['type'] != 'SUCCESS':
283            return False, result_obj['results'][0]['type']
284        return True, "SUCCESS"

302    def __update_body__(self) -> None:
303        if (self.sender is not None and self.recipient is not None and
304                self.target_timestamp is not None and self.target_author is not None and
305                self.recipient_type is not None):
306            # Removed reaction:
307            if self.is_remove:
308                if self.recipient_type == RecipientTypes.CONTACT:
309                    self.body = (f"{self.sender.get_display_name()} removed the "
310                                 f"reaction {self.emoji} "
311                                 f"from {self.target_author.get_display_name()}'s "
312                                 f"message {self.target_timestamp.get_timestamp()}.")
313                elif self.recipient_type == RecipientTypes.GROUP:
314                    self.body = (f"{self.sender.get_display_name()} removed the "
315                                 f"reaction {self.emoji} "
316                                 f"from {self.target_author.get_display_name()}'s "
317                                 f"message {self.target_timestamp.get_timestamp()} "
318                                 f"in group {self.recipient.get_display_name()}")
319                else:
320                    raise ValueError(f"recipient_type invalid value: {str(self.recipient_type)}")
321            # Changed reaction:
322            elif self.is_change:
323                if self.recipient_type == RecipientTypes.CONTACT:
324                    self.body = (f"{self.sender.get_display_name()} changed their "
325                                 f"reaction to {self.target_author.get_display_name()}'s "
326                                 f"message {self.target_timestamp.get_timestamp()}, "
327                                 f"from {self.previous_emoji} to {self.emoji}")
328                elif self.recipient_type == RecipientTypes.GROUP:
329                    self.body = (f"{self.sender.get_display_name()} changed their reaction "
330                                 f"to {self.target_author.get_display_name()}'s "
331                                 f"message {self.target_timestamp.get_timestamp()} in "
332                                 f"group {self.recipient.get_display_name()}, "
333                                 f"from {self.previous_emoji} to {self.emoji}")
334                else:
335                    raise ValueError(f"recipient_type invalid value: {str(self.recipient_type)}")
336            else:
337                # Added new reaction:
338                if self.recipient_type == RecipientTypes.CONTACT:
339                    self.body = (f"{self.sender.get_display_name()} reacted "
340                                 f"to {self.target_author.get_display_name()}'s message "
341                                 f"with {self.emoji}")
342                elif self.recipient_type == RecipientTypes.GROUP:
343                    self.body = (f"{self.sender.get_display_name()} reacted "
344                                 f"to {self.target_author.get_display_name()}'s "
345                                 f"message {self.target_timestamp.get_timestamp()} in "
346                                 f"group {self.recipient.get_display_name()} "
347                                 f"with {self.emoji}")
348                else:
349                    raise ValueError(f"recipient_type invalid value: {str(self.recipient_type)}")
350        else:
351            self.body = 'Invalid reaction.'
