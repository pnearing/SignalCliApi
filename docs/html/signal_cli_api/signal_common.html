104def valid_message_filter(message_filter: int) -> bool:
105    """
106    Evaluate a filter and determine if it's valid.
107    :param message_filter: int: The filter.
108    :return: bool: True the fileter if valid, False it is not.
109    """
110    max_value: int = (
111            MessageFilter.READ | MessageFilter.NOT_READ |
112            MessageFilter.VIEWED | MessageFilter.NOT_VIEWED |
113            MessageFilter.DELIVERED | MessageFilter.NOT_DELIVERED
114    )
115    if (message_filter < 0) or (message_filter > max_value):
116        return False
117    if (message_filter & MessageFilter.READ) and (message_filter & MessageFilter.NOT_READ):
118        return False
119    if (message_filter & MessageFilter.VIEWED) and (message_filter & MessageFilter.NOT_VIEWED):
120        return False
121    if (message_filter & MessageFilter.DELIVERED) and (message_filter &
122                                                       MessageFilter.NOT_DELIVERED):
123        return False
124    return True

360def __parse_signal_return_code__(return_code: int, command_line: str | list[str], output: str
361                                 ) -> NoReturn:
362    """
363    Parse the signal return code.
364    :param return_code: int: The return code from signal.
365    :param command_line: str | list[str]: The command line used to call signal.
366    :param output: str: The output generated by signal.
367    :raises SignalError: All the time with the error message and exit code.
368    """
369    logger_name = __name__ + '.' + __parse_signal_return_code__.__name__
370    logger: logging.Logger = logging.getLogger(logger_name)
371    logger.error("signal-cli returned non-zero return code: %i", return_code)
372    if return_code == 1:
373        error_message = f"Exit code 1: Invalid command line: {str(command_line)}"
374        logger.critical("Raising SignalError(%s).", error_message)
375        raise SignalError(error_message, return_code)
376    if return_code == 2:
377        error_message = f"Exit Code 2: Unexpected error. {output}"
378        logger.critical("Raising SignalError(%s).", error_message)
379        raise SignalError(error_message, return_code)
380    if return_code == 3:
381        error_message = f"Exit Code 3: Server / Network error. Try again later: {output}"
382        logger.critical("Raising SignalError(%s).", error_message)
383        raise SignalError(error_message, return_code)
384    if return_code == 4:
385        error_message = f"Exit Code 4: Operation failed due to untrusted key: {output}"
386        logger.critical("Raising SignalError(%s).", error_message)
387        raise SignalError(error_message, return_code)
388
389    error_message = (f"Exit Code {return_code}: Unknown / unhandled error. Running "
390                     f"'{str(command_line)}' returned output: {output}")
391    logger.critical("Raising SignalError(%s).", error_message)
392    raise SignalError(error_message, return_code)

406def __socket_create__(server_address: Optional[tuple[str, int] | str] = None) -> socket.socket:
407    """
408    Create a socket.socket object based on the server address type.
409    :param server_address: tuple[str, str] | str: The server address, either (HOSTNAME, PORT) or
410        "PATH_TO_SOCKET".
411    :return: socket.socket: The created socket.
412    :raises CommunicationsError: On failure to create socket.
413    """
414    if server_address is None:
415        server_address = SERVER_ADDRESS
416
417    if server_address is None:
418        raise ValueError("No server address defined.")
419
420    logger_name: str = __name__ + '.' + __socket_create__.__name__
421    logger: logging.Logger = logging.getLogger(logger_name)
422
423    if isinstance(server_address, (tuple, list)):
424        logger.debug("Creating INET socket.")
425        try:
426            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
427        except socket.error as e:
428            error_message: str = f"Failed to create INET socket: {str(e.args)}"
429            logger.critical("Raising CommunicationsError(%s).", error_message)
430            raise CommunicationsError(error_message, e) from e
431    elif isinstance(server_address, str):
432        logger.debug("Creating UNIX socket.")
433        try:
434            sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
435        except socket.error as e:
436            error_message: str = f"Failed to create UNIX socket: {str(e.args)}"
437            logger.critical("Raising CommunicationsError(%s).", error_message)
438            raise CommunicationsError(error_message, e) from e
439    else:
440        logger.critical("Raising TypeError:")
441        logger.critical(__type_err_msg__('server_address', 'tuple[str, int] | str', server_address))
442        __type_error__('server_address', 'tuple[str, int] | str', server_address)
443    # Create the socket dict and append it to the socket list.
444    socket_dict: dict[str, socket.socket | str | tuple[str, int]] = {
445        'server': server_address,
446        'socket': sock,
447        'status': 'created'
448    }
449    _OPEN_SOCKETS.append(socket_dict)
450    return sock

453def __socket_connect__(sock: socket.socket, server_address: Optional[tuple[str, int] | str] = None
454                       ) -> None:
455    """
456    Connect a socket to a server address.
457    :param sock: socket.socket: The socket to connect with.
458    :param server_address: tuple[str, int] | str: The server address: (HOSTNAME, PORT) or
459        "PATH_TO_SOCKET"
460    :return: None
461    :raises CommunicationsError: On failure to connect.
462    """
463    if server_address is None:
464        server_address = SERVER_ADDRESS
465    if server_address is None:
466        raise ValueError("Server address not defined.")
467
468    logger_name: str = __name__ + '.' + __socket_connect__.__name__
469    logger: logging.Logger = logging.getLogger(logger_name)
470    try:
471        # logger.debug("Connecting to: %s" % str(server_address))
472        # sock.connect(server_address)
473        logger.debug("Connecting to: %s", str(server_address))
474        sock.connect(server_address)
475    except socket.error as e:
476        error_message = f"Couldn't connect to socket: {str(e.args)}"
477        logger.critical("socket.error: %s, Raising CommunicationsError.", error_message)
478        raise CommunicationsError(error_message, e) from e
479    socket_dict = __find_socket_dict_by_socket__(sock)
480    socket_dict['status'] = 'connected'
481    logger.debug("Connected to: %s", str(server_address))

484def __socket_reconnect__(sock: socket.socket) -> None:
485    logger: logging.Logger = logging.getLogger(__name__ + '.' + __socket_reconnect__.__name__)
486    socket_dict = __find_socket_dict_by_socket__(sock)
487    if socket_dict['status'] == 'connected':
488        try:
489            __socket_close__(sock)
490        except socket.error:
491            warning_message = "Error while closing socket, ignoring."
492            logger.warning(warning_message)
493        try:
494            __socket_connect__(sock, socket_dict['server'])
495        except socket.error as e:
496            error_message = f"Couldn't reconnect the socket: {str(e.args)}"
497            logger.critical("socket.error: %s, raising CommunicationsError.", error_message)
498            raise CommunicationsError(error_message, e) from e

501def __socket_send__(sock: socket.socket, message: str) -> int:
502    """
503    Send a message to the socket.
504    :param sock: socket.socket: The socket to send the message over.
505    :param message: str: The message to send.
506    :return: int: The number of bytes sent.
507    :raises CommunicationsError: On failure to send.
508    """
509    logger_name: str = __name__ + '.' + __socket_send__.__name__
510    logger: logging.Logger = logging.getLogger(logger_name)
511    try:
512        logger.debug("Sending message: %s", message)
513        bytes_sent = sock.send(message.encode())
514    except socket.error as e:
515        if e.args[0] == 32:
516            logger.warning("Error while sending message. Broken pipe. Reconnecting.")
517            __socket_reconnect__(sock)
518            try:
519                logger.debug("Resending message: %s", message)
520                bytes_sent = sock.send(message.encode())
521            except socket.error as err:
522                error_message = f"Sending still failed: {str(e.args)}"
523                logger.critical("resending message failed: %s, raising CommunicationsError.",
524                                error_message)
525                raise CommunicationsError(error_message, e) from err
526        else:
527            error_message = f"Couldn't send to socket: {str(e.args)}"
528            logger.critical("socket.error: %s, Raising CommunicationsError.", error_message)
529            raise CommunicationsError(error_message, e) from e
530    logger.debug("Sent %i bytes.", bytes_sent)
531    return bytes_sent

534def __socket_receive_blocking__(sock: socket.socket) -> str:
535    """
536    Read a string from a socket; Blocks until msg read.
537    :param sock: socket.socket: The socket to read from.
538    :return: str: The read message.
539    :raises CommunicationsError: On failure to read from the socket.
540    """
541    logger_name: str = __name__ + '.' + __socket_receive_blocking__.__name__
542    logger: logging.Logger = logging.getLogger(logger_name)
543    try:
544        while True:
545            readable, _, erred = select.select([sock], [], [sock], 0.5)
546            if len(erred) > 0:
547                logger.critical("GOT ERRORS DURING SELECT.")
548            if len(readable) > 0:
549                message = b''
550                byte_count: int = 0
551                while True:
552                    data = sock.recv(1)
553                    message += data
554                    byte_count += 1
555                    try:
556                        if data.decode() == '\n':
557                            logger.debug("Received %i bytes.", byte_count)
558                            break
559                    except UnicodeDecodeError:
560                        pass
561                logger.debug("Returning message: %s", message.decode())
562                return message.decode()
563    except socket.error as e:
564        error_message = f"Failed to read from socket: {str(e.args)}"
565        if _CLOSING_SOCKET and e.args[0] == 9:
566            logger.info("Read error received while closing socket. This is normal during shutdown.")
567        else:
568            logger.critical("socket.error: %s", error_message)
569        raise CommunicationsError(error_message, e) from e

572def __socket_receive_non_blocking__(sock: socket.socket, wait_time: float = 0.1) -> Optional[str]:
573    """
574    Read a string from a socket; Blocks until msg read.
575    :param sock: socket.socket: The socket to read from.
576    :param wait_time: float: The waiting time in seconds for a message to appear.
577    :return: Optional[str]: The read message, or None if wait_time elapsed.
578    :raises CommunicationsError: On failure to read from the socket.
579    """
580    logger_name: str = __name__ + '.' + __socket_receive_non_blocking__.__name__
581    logger: logging.Logger = logging.getLogger(logger_name)
582    try:
583        readable, _, erred = select.select([sock], [], [sock], wait_time)
584        if len(erred) > 0:
585            logger.critical("GOT ERRORS WHILE SELECTING SOCKET.")
586        if len(readable) > 0:
587            message = b''
588            byte_count: int = 0
589            while True:
590                data = sock.recv(1)
591                message += data
592                byte_count += 1
593                try:
594                    if data.decode() == '\n':
595                        logger.debug("Received %i bytes.", byte_count)
596                        break
597                except UnicodeDecodeError:
598                    pass
599            logger.debug("Returning message: %s", message.decode())
600            return message.decode()
601    except socket.error as e:
602        error_message = f"Failed to read from socket: {str(e.args)}"
603        if _CLOSING_SOCKET:
604            logger.warning("Socket read error while closing socket."
605                           "This is normal during shutdown.")
606        else:
607            logger.critical("socket.error: %s", error_message)
608        raise CommunicationsError(error_message, e) from e
609    return None

653                                 non_fatal_errors: list[int] | tuple[int, ...] = (),
654                                 ) -> tuple[bool, int, str] | NoReturn:
655    """
656    Check the signal response object for error.
657    :param response_obj: dict[str, Any]: The response object from signal.
658    :param non_fatal_errors: list[int]: Any non-fatal error codes.
659    :return: tuple[bool, int, str] | NoReturn: If no error occurs, returns the tuple:
660        (False, 0, 'no error'); If a non-fatal error occurs, returns the tuple
661        (True, error_code: int, error_message: str); If a fatal error occurs then SignalError is
662        raised.
663    :raises: SignalError: On fatal error.
664    """
665    logger: logging.Logger = logging.getLogger(__name__ + '.' +
666                                               __check_response_for_error__.__name__)
667    if 'error' in response_obj.keys():
668        error: dict[str, Any] = response_obj['error']
669        if error['code'] in non_fatal_errors:
670            warning_message: str = (f"Signal error, code: {error['code']}, "
671                                    f"message: {error['message']}")
672            logger.warning(warning_message)
673            return True, error['code'], error['message']
674
675        error_message: str = f"Signal error, code: {error['code']}, message: {error['message']}"
676        logger.error("Raising SignalError(%s)", error_message)
677        raise SignalError(response_obj['error']['message'], response_obj['error']['code'],
678                          error_message)
679    return False, 0, 'no error'
